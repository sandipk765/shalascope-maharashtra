def check_file(path):

    if not path.exists():
        raise FileNotFoundError(
            f"Required input file not found:\n{path}"
        )


def load_csv(path):

    check_file(path)

    return pd.read_csv(
        path,
        low_memory=False
    )


# ============================================================
# 3. NUMERIC CONSTRAINTS
# ============================================================

def apply_bounds(values, low, high):

    values = np.asarray(
        values,
        dtype=float
    )

    values = np.where(
        np.isfinite(values),
        values,
        np.nan
    )

    if high is None:
        return np.maximum(values, low)

    return np.clip(
        values,
        low,
        high
    )

def select_models(summary):

    required = [
        "prediction",
        "model",
        "mae",
        "skill_vs_persistence",
        "validation_folds_won_vs_persistence",
        "total_validation_folds"
    ]

    missing = [
        c for c in required
        if c not in summary.columns
    ]

    if missing:
        raise ValueError(
            f"Model summary missing columns: {missing}"
        )

    summary = summary.copy()

    for column in [
        "mae",
        "skill_vs_persistence",
        "validation_folds_won_vs_persistence",
        "total_validation_folds"
    ]:
        summary[column] = pd.to_numeric(
            summary[column],
            errors="coerce"
        )

    selected = []

    for task in TASKS:

        group = summary[
            summary["prediction"] == task
        ].copy()

        if group.empty:
            raise ValueError(
                f"No model evaluation results for {task}"
            )

        persistence = group[
            group["model"] == "persistence"
        ]

        if len(persistence) != 1:
            raise ValueError(
                f"Expected one Persistence row for {task}"
            )

        # Conservative selection criterion:
        # positive skill against persistence and >=4/5 folds won.
        eligible = group[
            (group["model"] != "persistence")
            & (group["model"] != "district_historical_mean")
            & (group["skill_vs_persistence"] > 0)
            & (
                group["validation_folds_won_vs_persistence"] >= 4
            )
            & (group["total_validation_folds"] == 5)
        ].copy()

        if eligible.empty:

            winner = persistence.iloc[0]
            reason = (
                "No complex model met the positive-skill "
                "and four-of-five-fold stability rule; "
                "Persistence selected."
            )

        else:

            winner = eligible.sort_values(
                ["mae", "skill_vs_persistence"],
                ascending=[True, False]
            ).iloc[0]

            reason = (
                "Positive MAE skill and at least "
                "four validation folds won against Persistence."
            )

        selected.append({
            "prediction": task,
            "selected_model": winner["model"],
            "selected_mae": winner["mae"],
            "skill_vs_persistence": winner["skill_vs_persistence"],
            "validation_folds_won": int(
                winner["validation_folds_won_vs_persistence"]
            ),
            "total_validation_folds": int(
                winner["total_validation_folds"]
            ),
            "selection_reason": reason
        })

    result = pd.DataFrame(selected)

    result.to_csv(
        OUT / "forecast_model_selection_2026_27.csv",
        index=False,
        encoding="utf-8-sig"
    )

    return dict(
        zip(
            result["prediction"],
            result["selected_model"]
        )
    ), result

# 5. EMPIRICAL HISTORICAL ERROR RANGES
# ============================================================

def build_error_ranges(oof, selected_models):

    required = [
        "prediction",
        "model",
        "residual",
        "actual",
        "predicted",
        "validation_target_year"
    ]

    missing = [
        c for c in required
        if c not in oof.columns
    ]

    if missing:
        raise ValueError(
            f"OOF file missing columns: {missing}"
        )

    error_info = {}
    rows = []

    for task, config in TASKS.items():

        selected_model = selected_models[task]

        subset = oof[
            (oof["prediction"] == task)
            & (oof["model"] == selected_model)
        ].copy()

        subset["residual"] = pd.to_numeric(
            subset["residual"],
            errors="coerce"
        )

        residuals = subset["residual"].dropna()

        if len(residuals) < 30:
            raise ValueError(
                f"Insufficient historical OOF residuals "
                f"for {task}: {len(residuals)}"
            )

        q10 = float(residuals.quantile(0.10))
        q50 = float(residuals.quantile(0.50))
        q90 = float(residuals.quantile(0.90))

        error_info[task] = {
            "q10": q10,
            "q50": q50,
            "q90": q90,
            "count": len(residuals)
        }

        rows.append({
            "prediction": task,
            "selected_model": selected_model,
            "oof_residual_records": len(residuals),
            "residual_p10": q10,
            "residual_p50": q50,
            "residual_p90": q90,
            "unit": config["unit"],
            "method": (
                "Historical signed OOF residual P10-P90. "
                "Empirical error range, not a formal "
                "or guaranteed prediction interval."
            )
        })

    report = pd.DataFrame(rows)

    report.to_csv(
        OUT / "forecast_empirical_error_ranges_2026_27.csv",
        index=False,
        encoding="utf-8-sig"
    )

    return error_info

# 6. DISTRICT LINEAR TREND FOR DIGITAL READINESS
# ============================================================

def forecast_digital_trend(
    pairs,
    forecast_year,
    district_list
):

    required = [
        "district",
        "target_year_start",
        "target_next_year"
    ]

    missing = [
        c for c in required
        if c not in pairs.columns
    ]

    if missing:
        raise ValueError(
            f"Digital pair file missing columns: {missing}"
        )

    pairs = pairs.copy()

    pairs["district"] = (
        pairs["district"]
        .astype("string")
        .str.strip()
    )

    pairs["target_year_start"] = pd.to_numeric(
        pairs["target_year_start"],
        errors="coerce"
    )

    pairs["target_next_year"] = pd.to_numeric(
        pairs["target_next_year"],
        errors="coerce"
    )

    pairs = pairs.dropna(
        subset=[
            "district",
            "target_year_start",
            "target_next_year"
        ]
    )

    results = []

    for district in district_list:

        history = pairs[
            pairs["district"] == district
        ].copy()

        history = history.groupby(
            "target_year_start",
            as_index=False
        )["target_next_year"].mean()

        history = history.sort_values(
            "target_year_start"
        )

        if (
            len(history) < 3
            or history["target_year_start"].nunique() < 3
        ):
            raise ValueError(
                f"Insufficient digital trend history "
                f"for district: {district}"
            )

        x = history[
            "target_year_start"
        ].to_numpy(dtype=float)

        y = history[
            "target_next_year"
        ].to_numpy(dtype=float)

        slope, intercept = np.polyfit(
            x,
            y,
            1
        )

        prediction = (
            slope * float(forecast_year)
            + intercept
        )

        prediction = max(
            0.0,
            float(prediction)
        )

        results.append({
            "district": district,
            "forecast_digital_devices_per_school": prediction,
            "digital_trend_slope": float(slope),
            "digital_trend_history_years": len(history),
            "digital_trend_first_target_year": int(
                history["target_year_start"].min()
            ),
            "digital_trend_last_target_year": int(
                history["target_year_start"].max()
            )
        })

    return pd.DataFrame(results)

# 7. LOAD FROZEN NORMALIZATION PARAMETERS
# ============================================================

def load_normalization_parameters():

    params = load_csv(
        NORMALIZATION_INPUT
    )

    required = [
        "indicator",
        "direction",
        "p05",
        "p95"
    ]

    missing = [
        c for c in required
        if c not in params.columns
    ]

    if missing:
        raise ValueError(
            f"Normalization parameter columns missing: {missing}"
        )

    params = params.copy()

    params["p05"] = pd.to_numeric(
        params["p05"],
        errors="coerce"
    )

    params["p95"] = pd.to_numeric(
        params["p95"],
        errors="coerce"
    )

    required_indicators = [
        "Students per teacher",
        "Functional toilet percentage",
        "Good-condition classroom percentage",
        "Digital devices per reporting school"
    ]

    result = {}

    for indicator in required_indicators:

        match = params[
            params["indicator"] == indicator
        ]

        if len(match) != 1:
            raise ValueError(
                f"Expected exactly one normalization row "
                f"for: {indicator}"
            )

        row = match.iloc[0]

        if (
            pd.isna(row["p05"])
            or pd.isna(row["p95"])
            or row["p95"] <= row["p05"]
        ):
            raise ValueError(
                f"Invalid P5/P95 parameters for {indicator}"
            )

        result[indicator] = {
            "p05": float(row["p05"]),
            "p95": float(row["p95"]),
            "direction": str(row["direction"])
        }

    return result


# ============================================================
# 8. NORMALIZE FORECASTED INDICATORS
# ============================================================

def calculate_normalized_score(
    values,
    parameters
):

    values = pd.to_numeric(
        values,
        errors="coerce"
    )

    low = parameters["p05"]
    high = parameters["p95"]
    direction = parameters["direction"]

    clipped = values.clip(
        lower=low,
        upper=high
    )

    score = (
        (clipped - low)
        / (high - low)
        * 100
    )

    if direction == "LOWER_IS_BETTER":
        score = 100 - score

    elif direction != "HIGHER_IS_BETTER":
        raise ValueError(
            f"Unknown direction in normalization: {direction}"
        )

    return score.clip(
        lower=0,
        upper=100
    )


# ============================================================
# 9. HISTORICAL RISK REFERENCE THRESHOLDS
# ============================================================

def get_historical_thresholds(historical_scores):

    values = pd.to_numeric(
        historical_scores,
        errors="coerce"
    ).dropna()

    if len(values) != 288:
        raise ValueError(
            f"Expected 288 historical Shala Scores; got {len(values)}"
        )

    return {
        "q25": float(values.quantile(0.25)),
        "q50": float(values.quantile(0.50)),
        "q75": float(values.quantile(0.75))
    }


def classify_relative_band(score, thresholds):

    if pd.isna(score):
        return "NOT SCORED"

    if score <= thresholds["q25"]:
        return "CRITICAL"

    if score <= thresholds["q50"]:
        return "HIGH"

    if score <= thresholds["q75"]:
        return "MODERATE"

    return "LOW"


# ============================================================
# 10. APPLY HISTORICAL ERROR RANGE TO FORECAST
# ============================================================

def add_error_range(
    output,
    task,
    config,
    error_info
):

    prediction = pd.to_numeric(
        output[config["output_column"]],
        errors="coerce"
    )

    info = error_info[task]

    lower = prediction + info["q10"]
    upper = prediction + info["q90"]

    lower = apply_bounds(
        lower,
        config["lower_bound"],
        config["upper_bound"]
    )

    upper = apply_bounds(
        upper,
        config["lower_bound"],
        config["upper_bound"]
    )

    output[config["lower_column"]] = lower
    output[config["upper_column"]] = upper

    return output

def main():

    print("=" * 70)
    print("SHALASCOPE - 2026-27 FORECAST ENGINE")
    print("=" * 70)

    # --------------------------------------------------------
    # A. LOAD INPUT FILES
    # --------------------------------------------------------

    historical = load_csv(
        SCORE_INPUT
    )

    model_summary = load_csv(
        MODEL_SUMMARY_INPUT
    )

    oof = load_csv(
        OOF_INPUT
    )

    digital_pairs = load_csv(
        DIGITAL_PAIRS_INPUT
    )

    normalization = load_normalization_parameters()

    # --------------------------------------------------------
    # B. VALIDATE HISTORICAL DATA
    # --------------------------------------------------------

    required_columns = [
        "academic_year",
        "district",
        "school_count",
        "enrollment_reported_student_total",
        "score_teacher_total_derived",
        "functional_toilet_pct_raw",
        "good_classroom_pct_raw",
        "digital_devices_per_school_raw",
        "shala_score_v1"
    ]

    missing = [
        c for c in required_columns
        if c not in historical.columns
    ]

    if missing:
        raise ValueError(
            f"Historical dataset missing columns: {missing}"
        )

    historical = historical.copy()

    historical["academic_year"] = (
        historical["academic_year"].astype(str)
    )

    historical["year_start"] = pd.to_numeric(
        historical["academic_year"].str.extract(
            r"^(\d{4})"
        )[0],
        errors="coerce"
    )

    if historical["year_start"].isna().any():
        raise ValueError("Invalid academic year format")

    if len(historical) != 288:
        raise ValueError(
            f"Expected 288 historical rows; found {len(historical)}"
        )

    if historical["district"].nunique() != 36:
        raise ValueError("Expected 36 unique districts")

    if historical.duplicated(
        ["district", "year_start"]
    ).any():
        raise ValueError("Duplicate district-year records detected")

    latest_start = int(
        historical["year_start"].max()
    )

    if latest_start != 2025:
        raise ValueError(
            f"Expected latest historical year 2025-26; "
            f"found start year {latest_start}"
        )

    source_year = "2025-26"
    forecast_start = latest_start + 1

    forecast_year = (
        f"{forecast_start}-{str(forecast_start + 1)[-2:]}"
    )

    latest = historical[
        historical["year_start"] == latest_start
    ].copy()

    if len(latest) != 36:
        raise ValueError(
            f"Expected 36 latest-year districts; found {len(latest)}"
        )

    if latest["district"].nunique() != 36:
        raise ValueError(
            "Latest-year district count is not 36"
        )

    print("\nINPUT VALIDATION: PASS")
    print("Historical rows:", len(historical))
    print("Latest source year:", source_year)
    print("Forecast year:", forecast_year)
    print("Districts:", len(latest))

    # Convert forecast source fields to numeric.
    numeric_source_columns = [
        "school_count",
        "enrollment_reported_student_total",
        "score_teacher_total_derived",
        "functional_toilet_pct_raw",
        "good_classroom_pct_raw",
        "digital_devices_per_school_raw"
    ]

    for column in numeric_source_columns:
        latest[column] = pd.to_numeric(
            latest[column],
            errors="coerce"
        )

    # --------------------------------------------------------
    # C. SELECT MODELS
    # --------------------------------------------------------

    selected_models, selection_report = select_models(
        model_summary
    )

    print("\nSELECTED MODELS:")
    print(
        selection_report[
            [
                "prediction",
                "selected_model",
                "selected_mae",
                "skill_vs_persistence",
                "validation_folds_won"
            ]
        ].to_string(index=False)
    )

    # --------------------------------------------------------
    # D. HISTORICAL OOF ERROR RANGES
    # --------------------------------------------------------

    error_info = build_error_ranges(
        oof,
        selected_models
    )

    # --------------------------------------------------------
    # E. INITIALIZE DISTRICT OUTPUT
    # --------------------------------------------------------

    output = latest[
        [
            "district",
            "school_count",
            "enrollment_reported_student_total",
            "score_teacher_total_derived",
            "functional_toilet_pct_raw",
            "good_classroom_pct_raw",
            "digital_devices_per_school_raw"

        "score_teacher_total_derived": "source_teachers_2025_26",

    output.insert(

        "forecast_academic_year",
        forecast_year
    )

    output.insert(
        1,
        "source_academic_year",
        source_year
    )

    # --------------------------------------------------------
    # F. PERSISTENCE FORECASTS
    # --------------------------------------------------------

    source_mapping = {
        "student_enrollment":
            "source_enrollment_2025_26",

        "teacher_forecast":
            "source_teachers_2025_26",

        "infrastructure_toilet":
            "source_functional_toilet_pct_2025_26",

        "infrastructure_classroom":
            "source_good_classroom_pct_2025_26"
    }

    for task, source_column in source_mapping.items():

        config = TASKS[task]

        if selected_models[task] != "persistence":
            raise ValueError(
                f"{task}: selected model is "
                f"{selected_models[task]}, but this engine "
                f"currently implements Persistence for this task."
            )

        output[config["output_column"]] = (
            pd.to_numeric(
                output[source_column],
                errors="coerce"
            )
        )

        print(
            f"Forecast created: {task} -> Persistence"
        )

    # --------------------------------------------------------
    # G. DIGITAL READINESS DISTRICT TREND FORECAST
    # --------------------------------------------------------

    digital_task = "digital_readiness"

    digital_model = selected_models[digital_task]

    if digital_model == "district_linear_trend":

        digital_forecast = forecast_digital_trend(

            forecast_year=forecast_start,
            district_list=output["district"].tolist()
        )

        output = output.merge(

            on="district",
            how="left",
            validate="one_to_one"
        )

        if output[
            "forecast_digital_devices_per_school"
        ].isna().any():
            raise ValueError(
                "Missing digital forecast values"
            )

    elif digital_model == "persistence":

        )

    else:
        raise ValueError(
            f"Unsupported selected digital model: {digital_model}"
        )

    print(
        "Forecast created: digital_readiness ->",
        digital_model
    )

    # --------------------------------------------------------
    # H. CHECK FORECAST VALUES
    # --------------------------------------------------------

    forecast_checks = [
        ("forecast_enrollment", 0, None),
        ("forecast_teachers", 0, None),
        ("forecast_functional_toilet_pct", 0, 100),
        ("forecast_good_classroom_pct", 0, 100),
        ("forecast_digital_devices_per_school", 0, None)
    ]

    for column, low, high in forecast_checks:

        values = pd.to_numeric(
            output[column],
            errors="coerce"
        )

        if values.isna().any():
            raise ValueError(
                f"Missing forecasts in {column}"
            )

        if (values < low).any():
            raise ValueError(
                f"Invalid negative forecast in {column}"
            )

        if high is not None and (values > high).any():
            raise ValueError(
                f"Invalid percentage forecast in {column}"
            )

    # --------------------------------------------------------
    # I. ADD HISTORICAL EMPIRICAL ERROR RANGES
    # --------------------------------------------------------

    task_order = [
        "student_enrollment",
        "teacher_forecast",
        "infrastructure_toilet",
        "infrastructure_classroom",
        "digital_readiness"
    ]

    for task in task_order:

        output = add_error_range(
            output,
            task,
            TASKS[task],
            error_info
        )

    print("\nEMPIRICAL ERROR RANGES: CREATED")

    # --------------------------------------------------------
    # J. FORECAST PUPIL-TEACHER RATIO
    # --------------------------------------------------------

    output["forecast_students_per_teacher"] = (
        output["forecast_enrollment"]
        / output["forecast_teachers"].where(
            output["forecast_teachers"] > 0
        )

def constrain_prediction(values, bounds):
    """
    Apply valid output limits to predictions.
    """

    values = np.asarray(values, dtype=float)

    low, high = bounds

    values = np.where(
        np.isfinite(values),
        values,
        low
    )

    if high is None:
        return np.maximum(values, low)

    return np.clip(values, low, high)

def build_pipeline(model_type):
    """
    Build leakage-safe preprocessing + ML pipeline.

    Imputation, scaling and encoding are fitted
    separately inside each training fold.
    """

    numeric_pipeline = Pipeline([
        (
            "imputer",
            SimpleImputer(strategy="median")
        ),
        (
            "scaler",
            StandardScaler()
        )
    ])

    categorical_pipeline = Pipeline([
        (
            "imputer",
            SimpleImputer(strategy="most_frequent")
        ),
        (
            "onehot",
            OneHotEncoder(handle_unknown="ignore")
        )
    ])

    preprocessor = ColumnTransformer([
        (
            "numeric",
            numeric_pipeline,
            NUMERIC_FEATURES
        ),
        (
            "categorical",
            categorical_pipeline,
            CATEGORICAL_FEATURES
        )
    ])

    if model_type == "ridge":

        estimator = Ridge(
            alpha=10.0
        )

    elif model_type == "random_forest":

        estimator = RandomForestRegressor(
            n_estimators=300,
            max_depth=4,
            min_samples_leaf=4,
            max_features=0.8,
            random_state=42,
            n_jobs=-1
        )

    else:
        raise ValueError(
            f"Unknown model type: {model_type}"
        )

    return Pipeline([
        (
            "preprocessor",
            preprocessor
        ),
        (
            "model",
            estimator
        )
    ])

# 5. DISTRICT LINEAR TREND MODEL
# ============================================================

def district_linear_trend(
    train,
    test,
    target_column,
    target_year_column,
    bounds
):
    """
    Forecast each district using its historical target trend.

    Only training-period target observations are used.
    The validation target year is used only as the
    future time coordinate for extrapolation.
    """

    predictions = []

    global_mean = float(
        train[target_column].mean()
    )

    for _, test_row in test.iterrows():

        district = test_row["district"]

        future_year = float(
            test_row[target_year_column]
        )

        history = train[
            train["district"] == district
        ][
            [
                target_year_column,
                target_column
            ]
        ].dropna().copy()

        history = (
            history.groupby(
                target_year_column,
                as_index=False
            )[target_column]
            .mean()
            .sort_values(target_year_column)
        )

        if (
            len(history) >= 2
            and history[target_year_column].nunique() >= 2
        ):

            x = history[target_year_column].to_numpy(
                dtype=float
            )

            y = history[target_column].to_numpy(
                dtype=float
            )

            slope, intercept = np.polyfit(
                x,
                y,
                deg=1
            )

            prediction = (
                slope * future_year + intercept
            )

        elif len(history) == 1:

            prediction = float(
                history[target_column].iloc[-1]
            )

        else:

            prediction = global_mean

        predictions.append(prediction)

    return constrain_prediction(
        predictions,
        bounds
    )

def calculate_metrics(actual, predicted):

    actual = np.asarray(
        actual,
        dtype=float
    )

    predicted = np.asarray(
        predicted,
        dtype=float
    )

    return {
        "mae": float(
            mean_absolute_error(
                actual,
                predicted
            )
        ),

        "rmse": float(
            np.sqrt(
                mean_squared_error(
                    actual,
                    predicted
                )
            )
        ),

        "r2": float(
            r2_score(
                actual,
                predicted
            )
        )
    }

def load_task_data(task, config):

    path = PAIR_DIR / config["file"]

    if not path.exists():
        raise FileNotFoundError(
            f"Training-pair file not found: {path}"
        )

    data = pd.read_csv(
        path,
        low_memory=False
    )

    required = [
        "district",
        "feature_year_start",
        "target_year_start",
        *FEATURES,
        config["target"],
        config["current_value"]
    ]

    missing = [
        c for c in set(required)
        if c not in data.columns
    ]

    if missing:
        raise ValueError(
            f"{task}: missing columns: {missing}"
        )

    data["district"] = (
        data["district"]
        .astype("string")
        .str.strip()
    )

    numeric_columns = (
        NUMERIC_FEATURES
        + [
            "target_year_start",
            config["target"],
            config["current_value"]
        ]
    )

    for column in numeric_columns:
        data[column] = pd.to_numeric(
            data[column],
            errors="coerce"
        )

    data = data.dropna(
        subset=[
            "district",
            "feature_year_start",
            "target_year_start",
            config["target"],
            config["current_value"]
        ]
    ).copy()

    data["feature_year_start"] = (
        data["feature_year_start"].astype(int)
    )

    data["target_year_start"] = (
        data["target_year_start"].astype(int)
    )

    # Ensure that the target is exactly one academic year ahead.
    valid_transition = (
        data["target_year_start"]
        == data["feature_year_start"] + 1
    )

    data = data[valid_transition].copy()

    if data.empty:
        raise ValueError(
            f"{task}: no valid consecutive-year pairs"
        )

    if data.duplicated(
        ["district", "target_year_start"]
    ).any():

        raise ValueError(
            f"{task}: duplicate district-target-year records"
        )

    return data.sort_values(
        ["target_year_start", "district"]
    ).reset_index(drop=True)

            )

        print("Available pairs:", len(data))
        print("Validation target years:", validation_years)

        # ====================================================
        # EXPANDING WINDOW VALIDATION
        # ====================================================

        for validation_year in validation_years:

            train = data[
                data["target_year_start"]
                < validation_year
            ].copy()

            test = data[
                data["target_year_start"]
                == validation_year
            ].copy()

            if len(train) == 0 or len(test) != 36:

                raise ValueError(
                    f"{task}: invalid validation fold "
                    f"for {validation_year}. "
                    f"Train={len(train)}, Test={len(test)}"
                )

            X_train = train[FEATURES]
            y_train = train[
                config["target"]
            ].astype(float)

            X_test = test[FEATURES]
            y_test = test[
                config["target"]
            ].astype(float)

            models = {}

            # ------------------------------------------------
            # MODEL 1: PERSISTENCE
            # ------------------------------------------------

            persistence_raw = test[
                config["current_value"]
            ].to_numpy(dtype=float)

            models["persistence"] = (
                constrain_prediction(
                    persistence_raw,
                    config["bounds"]
                )
            )

            # ------------------------------------------------
            # MODEL 2: DISTRICT HISTORICAL MEAN
            # ------------------------------------------------

            district_mean = (
                train.groupby("district")[
                    config["target"]
                ].mean()
            )

            global_mean = float(
                y_train.mean()
            )

            mean_raw = (
                test["district"]
                .map(district_mean)
                .fillna(global_mean)
                .to_numpy(dtype=float)
            )

            models["district_historical_mean"] = (
                constrain_prediction(
                    mean_raw,
                    config["bounds"]
                )
            )

            # ------------------------------------------------
            # MODEL 3: DISTRICT LINEAR TREND
            # ------------------------------------------------

            trend_predictions = district_linear_trend(
                train=train,
                test=test,
                target_column=config["target"],
                target_year_column="target_year_start",
                bounds=config["bounds"]
            )

            models["district_linear_trend"] = (
                trend_predictions
            )

            # ------------------------------------------------
            # MODEL 4: RIDGE REGRESSION
            # ------------------------------------------------

            ridge = build_pipeline(
                "ridge"
            )

            with warnings.catch_warnings():
                warnings.simplefilter("ignore")
                ridge.fit(
                    X_train,
                    y_train

                X_test

                    "rmse": metrics["rmse"],
                    "r2": metrics["r2"]
                })

                for index, (_, row) in enumerate(
                    test.iterrows()

                    actual = float(
                        y_test.iloc[index]
                    )

                        predictions[index]
                    )

                    prediction_records.append({
                        "prediction": task,
                        "validation_target_year": validation_year,
                        "district": row["district"],
                        "model": model_name,
                        "actual": actual,
                        "predicted": predicted,
                        "residual": actual - predicted,
                        "absolute_error": abs(
                            actual - predicted
                        ),
                        "unit": config["unit"]
                    })

                print(
                    f"  {model_name:28s}"
                    f" MAE={metrics['mae']:.4f}"
                    f" RMSE={metrics['rmse']:.4f}"
                    f" R2={metrics['r2']:.4f}"
                )

    # ========================================================
    # 9. CREATE OUTPUT DATAFRAMES
    # ========================================================

    folds = pd.DataFrame(
        fold_records
    )

    predictions = pd.DataFrame(
        prediction_records
    )

    if folds.empty or predictions.empty:
        raise RuntimeError(
            "No validation results were generated"
        )

    # ========================================================
    # 10. SAVE FOLD METRICS
    # ========================================================

    folds.to_csv(
        OUT / "baseline_fold_metrics.csv",
        index=False,
        encoding="utf-8-sig"
    )

    # ========================================================
    # 11. SAVE OUT-OF-FOLD PREDICTIONS
    # ========================================================

    predictions.to_csv(
        OUT / "baseline_oof_predictions.csv",
        index=False,
        encoding="utf-8-sig"
    )

    # ========================================================
    # 12. MODEL COMPARISON SUMMARY
    # ========================================================

    summary_rows = []

    for task, task_predictions in predictions.groupby(
        "prediction"
    ):

        persistence = task_predictions[
            task_predictions["model"]
            == "persistence"
        ]

        persistence_mae = float(
            persistence["absolute_error"].mean()
        )

        persistence_by_fold = (
            folds[
                (folds["prediction"] == task)
                & (folds["model"] == "persistence")
            ][
                [
                    "validation_target_year",
                    "mae"
                ]
            ].rename(
                columns={
                    "mae": "persistence_fold_mae"
                }
            )
        )

        for model_name, model_predictions in (
            task_predictions.groupby("model")
        ):

            actual = model_predictions[
                "actual"
            ]

            predicted = model_predictions[
                "predicted"
            ]

            metrics = calculate_metrics(
                actual,
                predicted
            )

            model_by_fold = folds[
                (folds["prediction"] == task)
                & (folds["model"] == model_name)
            ]

            comparison = model_by_fold.merge(
                persistence_by_fold,
                on="validation_target_year",
                how="left"
            )

            wins = int(
                (
                    comparison["mae"]
                    < comparison["persistence_fold_mae"]
                ).sum()
            )

            skill = (
                1 - metrics["mae"] / persistence_mae
                if persistence_mae > 0
                else np.nan
            )

            summary_rows.append({
                "prediction": task,
                "model": model_name,
                "oof_predictions": len(model_predictions),
                "mae": metrics["mae"],
                "rmse": metrics["rmse"],
                "r2": metrics["r2"],
                "persistence_mae": persistence_mae,
                "skill_vs_persistence": skill,
                "validation_folds_won_vs_persistence": wins,
                "total_validation_folds": len(
                    comparison
                )
            })

    summary = pd.DataFrame(
        summary_rows
    )

    summary = summary.sort_values(
        ["prediction", "mae"]
    )

    summary.to_csv(
        OUT / "baseline_model_summary.csv",
        index=False,
        encoding="utf-8-sig"
    )

    # ========================================================
    # 13. BEST MODEL PER TASK
    # ========================================================

    best_models = (
        summary.sort_values(
            ["prediction", "mae"]
        )
        .groupby("prediction", as_index=False)
        .first()
    )

    best_models.to_csv(
        OUT / "baseline_best_model_by_task.csv",
        index=False,
        encoding="utf-8-sig"
    )

    # ========================================================
    # 14. METHODOLOGY REPORT
    # ========================================================

    methodology = [
        "SHALASCOPE - BASELINE ML EXPERIMENTS V2",
        "",
        "Validation: Expanding-window chronological validation.",
        "Validation target years: 2021-22 through 2025-26.",
        "Each validation year contains 36 districts.",
        "No random train-test split was used.",
        "",
        "Five model approaches:",
        "1. Persistence baseline.",
        "2. District historical mean.",
        "3. District linear trend.",
        "4. Ridge regression.",
        "5. Random Forest.",
        "",
        "Forecast features come from the preceding academic year.",
        "Forecast target is from the next consecutive academic year.",
        "Historical mean uses training target values only.",
        "District trend uses past training target observations only.",
        "Trend extrapolation uses the known validation year coordinate.",
        "ML preprocessing is fitted separately inside each training fold.",
        "",
        "Primary metric: MAE.",
        "Secondary metrics: RMSE and R2.",
        "Skill = 1 - model MAE / persistence MAE.",
        "Positive skill means improvement over persistence.",
        "Negative skill means worse MAE than persistence.",
        "Fold wins count validation periods with lower MAE than persistence.",
        "",
        "There are only five temporal validation folds.",
        "Repeated observations from each district are not independent.",
        "Model selection considers overall MAE and fold stability.",
        "No model is automatically production-ready.",
        "",
        "Future Shala Score must be derived from forecast raw indicators.",
        "It must not be trained directly using full-period normalized scores.",
        "Normalization for future scoring must use training-period parameters."
    ]

    (
        OUT / "baseline_experiment_methodology.txt"
    ).write_text(
        "\n".join(methodology),
        encoding="utf-8"
    )

    # ========================================================
    # 15. FINAL SUMMARY
    # ========================================================

    print("\n" + "=" * 70)
    print("SHALASCOPE - BASELINE ML EXPERIMENT SUMMARY")
    print("=" * 70)

    print(
        summary.to_string(index=False)
    )

    print("\nBEST MODEL BY PREDICTION:")

    print(
        best_models[
            [
                "prediction",
                "model",
                "mae",
                "skill_vs_persistence",
                "validation_folds_won_vs_persistence"
            ]
        ].to_string(index=False)
    )

    print("\nOUTPUT FILES:")
    print(OUT / "baseline_fold_metrics.csv")
    print(OUT / "baseline_oof_predictions.csv")
    print(OUT / "baseline_model_summary.csv")
    print(OUT / "baseline_best_model_by_task.csv")
    print(OUT / "baseline_experiment_methodology.txt")

    print("\nSTATUS: EXPERIMENT COMPLETED")
    print("No model has been declared production-ready.")


if __name__ == "__main__":
    main()

Digital readiness: District linear trend.

Persistence uses the latest 2025-26 observed value.
Digital trend uses historical target-year observations
available through 2025-26 and extrapolates to 2026-27.

Historical error ranges use pooled out-of-fold residual
P10 and P90 quantiles from 2021-22 to 2025-26.
These are empirical error ranges, not formal prediction
intervals and do not guarantee 80 percent coverage.

Future Shala Score is derived from forecast raw indicators.
Historical V1 P5-P95 parameters are frozen and reused.
Teacher score uses forecast students per teacher.
Infrastructure score is the mean of toilet and classroom scores.
Digital score uses forecast devices per reporting school.
All three pillars have equal weights.

Relative performance bands use historical V1 score quartiles.
They are descriptive categories, not risk probabilities.

Forecasts are experimental and not official government estimates.
No student learning outcome or causal impact is inferred.
